	function onFullNameChange(e) {
		const value = e.target.value.trim();
		setFullName(value);

		if (value == "")
			setFullNameLabelHTML({
				__html: '<span class = "text-red-500"> You cannot leave this field empty! Please enter your full name:</span>',
			});
		else
			setFullNameLabelHTML({

			});

			setCountryLabelHTML({

			});
	}
