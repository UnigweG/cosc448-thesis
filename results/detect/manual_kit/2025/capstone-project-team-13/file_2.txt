function normalizeContactInfo(raw) {
  const contact = { ...DEFAULT_CONTACT, ...(raw || {}) };
  contact.customPresences = Array.isArray(contact.customPresences)
    ? contact.customPresences.map((item, idx) => ({
        id: item?.id || `presence-${Date.now()}-${idx}`,
        label: item?.label || "",
        url: item?.url || "",
      }))
    : [];
  return contact;
}
