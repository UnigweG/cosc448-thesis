const getCategoryIcon = (category: string) => {
  switch (category) {
    case 'Food and Drink':
      return (
        <FontAwesomeIcon
          icon={faBowlFood}
          className="text-blue-500 inline-block mr-1"
        />
      );
    case 'Restaurants':
      return (
        <FontAwesomeIcon
          icon={faUtensils}
          className="text-grey-500 inline-block mr-1"
        />
      );
    case 'Recreation':
      return (
        <FontAwesomeIcon
          icon={faUmbrellaBeach}
          className="text-blue-500 inline-block mr-1"
        />
      );
    case 'Travel':
      return (
        <GlobeAmericasIcon className="h-4 w-4 text-yellow-500 inline-block mr-1" />
      );
    case 'Fast Food':
      return (
        <FontAwesomeIcon
          icon={faBurger}
          className="text-brown-500 inline-block mr-1"
        />
      );
    case 'Credit Card':
      return <CreditCardIcon className="h-4 w-4 inline-block mr-1" />;
    case 'Transfer':
      return (
        <FontAwesomeIcon
          icon={faMoneyBillTransfer}
          className="h-4 w-4 inline-block mr-1"
        />
      );
    case 'Airlines and Aviation Services':
      return (
        <FontAwesomeIcon icon={faPlane} className="h-4 w-4 inline-block mr-1" />
      );
    case 'Taxi':
      return (
        <FontAwesomeIcon icon={faTaxi} className="h-4 w-4 inline-block mr-1" />
      );
    default:
      return <ShoppingCartIcon className="h-4 w-4 inline-block mr-1" />;
  }
};
