export const GeneralSettings: React.FC = () => {
  const { userId } = useAuth();
  const { currencyCode, refreshAuthState } = useAuth();
  const { toast } = useToast();

  const [availableCurrencies, setAvailableCurrencies] = useState<string[]>([]);
  const [selectedCurrency, setSelectedCurrency] = useState<string>('');
  const [conversionRate, setConversionRate] = useState<number | null>(null);
  const [newCurrency, setNewCurrency] = useState<string>('');
  const [loading, setLoading] = useState<boolean>(false);
  const [error, setError] = useState<string | null>(null);
  const [success, setSuccess] = useState<string | null>(null);
  const [open, setOpen] = useState(false);

  // Accounting state
  const [accountingInfo, setAccountingInfo] = useState<AccountingInfo | null>(
    null
  );
  const [monthlyIncome, setMonthlyIncome] = useState<number | ''>('');
  const [isEditing, setIsEditing] = useState(false);
  const [accountingError, setAccountingError] = useState<string | null>(null);
  const [accountingSuccess, setAccountingSuccess] = useState<string | null>(
    null
  );
  const [accountingLoading, setAccountingLoading] = useState(false);
  const [isCalculating, setIsCalculating] = useState(false);

  // Fetch available currencies on mount
  useEffect(() => {
    const fetchCurrencies = async () => {
      try {
        const response = await fetch(
          `${process.env.BACKEND_URL}/api/currencies`
        );
        const data = await response.json();

        if (response.ok) {
          setAvailableCurrencies(
            data.map((currency: Currency) => currency.countryCode)
          );
        } else {
          console.error('Failed to fetch currencies:', data.error);
        }
      } catch (error) {
        console.error('Error fetching currencies:', error);
      }
    };

    fetchCurrencies();
  }, []);

  // Fetch the conversion rate between user's current currency and selected currency
  useEffect(() => {
    const fetchConversionRate = async () => {
      if (!selectedCurrency || !currencyCode) {
        setConversionRate(null);
        return;
      }

      try {
        const response = await fetch(
          `https://cors-anywhere.herokuapp.com/https://api.exchangerate-api.com/v4/latest/${currencyCode}`
        );
        const data = await response.json();

        if (response.ok && data.rates[selectedCurrency]) {
          setConversionRate(data.rates[selectedCurrency]);
        } else {
          console.error('Failed to fetch conversion rate:', data.error);
        }
      } catch (error) {
        console.error('Error fetching conversion rate:', error);
      }
    };

    fetchConversionRate();
  }, [selectedCurrency, currencyCode]);

  // Fetch accounting info on mount
  useEffect(() => {
    const fetchAccountingInfo = async () => {
      try {
        const response = await axios.get(
          `${process.env.BACKEND_URL}/api/accountingInfo/${userId}`
        );
        setAccountingInfo(response.data);
        setMonthlyIncome(response.data.monthlyIncome);
      } catch (err) {
        console.error(err);
        setAccountingError('Failed to fetch accounting information');
      }
    };

    if (userId) {
      fetchAccountingInfo();
    }
