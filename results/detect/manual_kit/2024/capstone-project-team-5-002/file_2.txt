def load_parameters(params_file):
    """Load and parse parameters file"""
    with open(params_file, 'r') as f:
        return json.load(f)

def load_3d_image_selections(json_file):
    """Load 3D image selection data from JSON file"""
    try:
        with open(json_file, 'r') as f:
            selections = json.load(f)
        return selections
    except Exception as e:
        print(f"Error loading 3D image selections data: {e}")
        return []

def extract_png_metadata(image_path):
    """Extract metadata from a PNG file using PIL"""
    try:
        with Image.open(image_path) as img:
            metadata = img.info
            
            # Parse any JSON strings in the metadata
            parsed_metadata = {}
            for key, value in metadata.items():
                if isinstance(value, str):
                    try:
                        if key in ['global_params', 'image_params', 'surfaces']:
                            parsed_metadata[key] = json.loads(value)
                        else:
                            parsed_metadata[key] = value
                    except json.JSONDecodeError:
                        parsed_metadata[key] = value
                else:
                    parsed_metadata[key] = value
                    
            return parsed_metadata
    except Exception as e:
        print(f"Error extracting metadata from {image_path}: {e}")
        return {}
