@dataclass
class SceneParameters:
    """Parameters for scene generation"""
    simplicity: int           # Number of shapes (1-20)
    expressiveness: float     # Distortion amount (0-1)
    size_uniformity: float    # Size variation (0-1)
    orientation_uniformity: float  # Rotation variation (0-1)
    symmetry_axis: int        # Symmetry axis (0-none, 1-x, 2-y, 3-z)
    asymmetric_objects: int   # Number of objects not in symmetry
    
    def to_dict(self):
        return {
            'simplicity': self.simplicity,
            'expressiveness': self.expressiveness,
            'size_uniformity': self.size_uniformity,
            'orientation_uniformity': self.orientation_uniformity,
            'symmetry_axis': self.symmetry_axis,
            'asymmetric_objects': self.asymmetric_objects
        }
    
    def generate_similar(self):
        """Generate slightly varied parameters"""
        return SceneParameters(
            simplicity=max(1, min(20, self.simplicity + np.random.randint(-1, 1))),
            expressiveness=max(0, min(1, self.expressiveness + np.random.normal(0, 0.01))),
            size_uniformity=max(0, min(1, self.size_uniformity + np.random.normal(0, 0.01))),
            orientation_uniformity=max(0, min(1, self.orientation_uniformity + np.random.normal(0, 0.01))),
            symmetry_axis=self.symmetry_axis,  # Keep same symmetry axis
            asymmetric_objects=max(0, min(self.simplicity, 
                                        self.asymmetric_objects + np.random.randint(-1, 1)))
        )
