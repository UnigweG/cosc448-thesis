-- Get percentage breakdown of answers for each question a specific user answered
SELECT 
    iqs.image_question_id,
    iqs.questionare_order,
    iq.img1_id,
    ROUND((COUNT(CASE WHEN iqs2.selected = iq.img1_id THEN 1 END) * 100.0 / COUNT(*)), 2) as img1_percentage,
    iq.img2_id,
    ROUND((COUNT(CASE WHEN iqs2.selected = iq.img2_id THEN 1 END) * 100.0 / COUNT(*)), 2) as img2_percentage,
    CASE WHEN iq.img3_id IS NOT NULL THEN iq.img3_id END as img3_id,
    CASE WHEN iq.img3_id IS NOT NULL THEN 
        ROUND((COUNT(CASE WHEN iqs2.selected = iq.img3_id THEN 1 END) * 100.0 / COUNT(*)), 2)
    END as img3_percentage
FROM image_question_selection iqs
JOIN image_question iq ON iqs.image_question_id = iq.id
JOIN image_question_selection iqs2 ON iqs.image_question_id = iqs2.image_question_id
WHERE iqs.participant_id = 48  -- Replace with desired participant_id
GROUP BY iqs.image_question_id, iqs.questionare_order, iq.img1_id, iq.img2_id, iq.img3_id
ORDER BY iqs.questionare_order ASC;

-- New table to store raw aesthetic parameters
CREATE TABLE image_analysis (
id INTEGER PRIMARY KEY AUTOINCREMENT,
set_number INTEGER NOT NULL,
image_question_id INTEGER NOT NULL,

-- Raw parameter values
symmetry_orientation INTEGER NOT NULL, -- 0-none, 1-horizontal, 2-vertical
num_shapes INTEGER NOT NULL, -- Higher means lower simplicity
size_variation FLOAT NOT NULL, -- 0 to 1
orientation_variation FLOAT NOT NULL, -- 0 to 1
shape_distortion FLOAT NOT NULL, -- 0 to infinity

liked_percentage FLOAT NOT NULL -- Percentage of users who selected this image
);
