function wrapColumnsInQuotes(sql: string, columns: string[]): string {
    return columns.reduce((acc, column) => {
        const regex = new RegExp(`(?<!")\\b${column}\\b(?!")`, "g");
        return acc.replace(regex, `"${column}"`);
    }, sql);
}

export async function generateUserSpecific(query: string, preferredCurrency: string, userId: string, convertedTransactions: Transaction[], convertedAccounts: Account[], chatHistory: {role: string;content: string;}[], doGenerateGraph: boolean): Promise<string> {
    try { 
        
        const embedding = await openAI.embeddings.create({
            model: "text-embedding-ada-002",
            input: query
        });

        const index = await pinecone.index(process.env.PINECONE_INDEX_NAME);

        const vectorQueryResults = await index.namespace("transactions").query({
            topK: 10,
            vector: embedding.data[0].embedding,
            includeMetadata: true,
        });

        const pineconeFilters = {
            merchants: new Set<string>(),
            categories: new Set<string>(),
            subcategories: new Set<string>(),
        };

        vectorQueryResults.matches?.forEach((match) => {
            const merchant = match.metadata?.merchantName;
            if (typeof merchant === "string") pineconeFilters.merchants.add(merchant);
          
            const category = match.metadata?.category;
            if (typeof category === "string") pineconeFilters.categories.add(category);
          
            const subCategory = match.metadata?.subCategory;
            if (typeof subCategory === "string") pineconeFilters.subcategories.add(subCategory);
        });          

        const pineconeContext = `
- **Relevant Filters from Pinecone Semantic Search**:
Merchants: ${Array.from(pineconeFilters.merchants).join(", ") || "None"}
Categories: ${Array.from(pineconeFilters.categories).join(", ") || "None"}
Subcategories: ${Array.from(pineconeFilters.subcategories).join(", ") || "None"}
        `;

        const sqlPrompt = `
            You are an expert in generating SQL queries for a Prisma-backed PostgreSQL database. Your goal is to create accurate, optimized queries that retrieve relevant data while adhering to the database schema and user query requirements.
        
            - **Database Schema**:
            ${prismaSchema}
            
            - **User Information**:
            User ID: ${userId}
            
            - **User Query**:
            "${query}"
            
            - **Chat History**:
            ${JSON.stringify(chatHistory, null, 2)}

            - **Transactions Context**:
            ${JSON.stringify(convertedTransactions, null, 2)}
