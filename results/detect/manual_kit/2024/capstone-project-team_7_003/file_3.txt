const base_path = process.env.BASE_PATH;

export default function Settings() {
  const { user, setUser, updateUserOnServer } = useUser();
  const [localUser, setLocalUser] = useState<Omit<User, 'password'>>({
    id: user?.id || '',
    email: user?.email || '',
    firstName: user?.firstName || '',
    lastName: user?.lastName || '',
    role: user?.role || 'USER',
    preferredCurrency: user?.preferredCurrency || 'USD',
    preferredTheme: user?.preferredTheme || 'Light',
  });
  const [newPassword, setNewPassword] = useState("");
  const [currentPassword, setCurrentPassword] = useState("");
  const [accounts, setAccounts] = useState<Account[]>([]);
  const [passwordVisible, setPasswordVisible] = useState<{
    currentPasswordVisible: boolean;
    newPasswordVisible: boolean;
  }>({ currentPasswordVisible: false, newPasswordVisible: false });
  const [passwordStrength, setPasswordStrength] = useState("");
  const [saveLoadingState, setSaveLoadingState] = useState(false);
  const [deleteLoadingState, setDeleteLoadingState] = useState(false);
  const [passwordChangeLoading, setPasswordChangeLoading] = useState(false);

  const notifySuccess = (message: string) => toast.success(message);
  const notifyError = (message: string) => toast.error(message);

  useEffect(() => {
    // Sync local state with context user
    if (user) {
      setLocalUser({
        id: user.id,
        email: user.email,
        firstName: user.firstName,
        lastName: user.lastName,
        role: user.role,
        preferredCurrency: user.preferredCurrency,
        preferredTheme: user.preferredTheme,
      });
    }
  }, [user]);

  useEffect(() => {
    const fetchAccounts = async () => {
      try {
        const response = await fetch(`${base_path}api/user/get-accounts`);

        if (!response.ok) {
          throw new Error("Failed to fetch users");
        }

        const data = await response.json();

        setAccounts([...data.accounts]);
      } catch (error) {
        console.error(error);
      }
    };

    fetchAccounts();
  }, []);

  const handlePreferredCurrencyChange = (currency: string) => {
    setLocalUser((prev) => ({ ...prev, preferredCurrency: currency }));
  };

  const handlePreferredThemeChange = (theme: string) => {
    setLocalUser((prev) => ({ ...prev, preferredTheme: theme }));
  };

  const handleSaveSettings = async () => {
    setSaveLoadingState(true);
    try {
      const response = await updateUserOnServer(localUser);

      if (!response.ok) {
        notifyError("Failed to update user!");
        throw new Error("Failed to update user.");
      }

      setUser(localUser); // Update context
      notifySuccess("Updated user successfully!");
    } catch (error) {
      notifyError("Failed to update user!");
      throw new Error("Failed to update user:" + error);
    } finally {
      setSaveLoadingState(false);
    }
  };

  const handleDeleteProfile = async () => {
    if (
      !confirm(
        "Are you sure you want to delete your account? This action cannot be undone."
      )
    ) {
      return;
    }

    setDeleteLoadingState(true);
    try {
      const response = await fetch(`${base_path}api/user/delete-account`, {
        method: "DELETE",
      });

      if (!response.ok) {
        throw new Error("Failed to delete account");
      }

      notifySuccess("Account deleted successfully!");
      window.location.href = base_path;
    } catch (error) {
      console.error("Error deleting account:", error);
      notifyError("Failed to delete account. Please try again.");
    } finally {
      setDeleteLoadingState(false);
    }
  };
