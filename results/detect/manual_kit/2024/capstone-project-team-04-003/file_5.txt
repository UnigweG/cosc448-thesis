// Add this function before the component class
function containsNaughtyWords(
  control: AbstractControl
): ValidationErrors | null {
  if (!control.value) return null;

  const name: string = control.value.toLowerCase();
  // Check both English and French naughty words
  const hasNaughtyWord: boolean =
    naughtyWords.en.some((word: string) => name.includes(word.toLowerCase())) ||
    naughtyWords.fr.some((word: string) => name.includes(word.toLowerCase()));

  return hasNaughtyWord ? { naughtyWord: true } : null;
}
