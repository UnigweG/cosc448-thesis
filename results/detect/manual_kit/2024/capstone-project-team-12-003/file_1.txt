// Initialize Image Grid
function initializeImageGrid() {
    const imageGrid = document.getElementById('imageGrid');

    // Add drag and drop area
    const dragDropArea = document.createElement('div');
    dragDropArea.className = 'drag-drop-area';
    dragDropArea.innerHTML = `
        <i class="fas fa-cloud-upload-alt" style="font-size: 24px; color: #78B97B;"></i>
        <p>Drag and drop images here or click to upload</p>
        <input type="file" multiple accept="image/*" style="display: none;" id="fileInput">
    `;

    // Image grid content
    const gridContent = document.createElement('div');
    gridContent.className = 'image-grid';
    imageGrid.appendChild(dragDropArea);
    imageGrid.appendChild(gridContent);

    dragDropArea.addEventListener('click', () => {
        document.getElementById('fileInput').click();
    });

    document.getElementById('fileInput').addEventListener('change', handleFileSelect);
    setupDragAndDrop(dragDropArea);
    fetchImages();
}

async function fetchImages() {
    try {
        const response = await fetch('https://localhost:3000/api/images');
        if (!response.ok) {
            throw new Error('Failed to fetch images');
        }

        const images = await response.json();
        console.log(images)
        const reversedImages = images.reverse();
        reversedImages.forEach(image => {
            addImageToGrid({
                id: image.image_id,
                path: image.url,
                name: `Image ${image.image_id}`
            });
        });
    } catch (error) {
        console.error('Error fetching images:', error);
    }
}
function showNotification(message, type = 'success') {
    // Create a notification element
    const notification = document.createElement('div');
    notification.className = `notification ${type}`;
    notification.innerText = message;

    // Add the notification to the body
    document.body.appendChild(notification);

    // Remove the notification after 3 seconds
    setTimeout(() => {
        notification.remove();
    }, 3000);
}

function setupDragAndDrop(dropArea) {
    ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
        dropArea.addEventListener(eventName, preventDefaults, false);
    });

    dropArea.addEventListener('drop', handleDrop);
    function preventDefaults(e) {
        e.preventDefault();
        e.stopPropagation();
    }
}

function handleDrop(e) {
    const dt = e.dataTransfer;
    const files = dt.files;
    handleFiles(files);
}

function handleFileSelect(e) {
    const files = e.target.files;
    handleFiles(files);
}

function handleFiles(files) {
    [...files].forEach(file => {
        if (file.type.startsWith('image/')) {
            uploadFile(file);
        }
    });
}
