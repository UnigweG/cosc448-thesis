document.body.innerHTML = `

// Import your functions here
import { initializeStats, initializeImageGrid, initializeAlgorithmStatus, initializeRecentComparisons, initializeResearchMetrics } from './dashboard';

// Test for initializeStats
describe('initializeStats', () => {
    test('renders stats correctly', () => {
        initializeStats();
        const statsGrid = document.getElementById('statsGrid');

        // Check each stat item in statsGrid
        dashboardData.stats.forEach(stat => {
            expect(statsGrid).toHaveTextContent(stat.label);
            expect(statsGrid).toHaveTextContent(stat.value);
        });
    });
});

// Test for initializeImageGrid
describe('initializeImageGrid', () => {
    test('renders image grid correctly with drag-and-drop area', () => {
        initializeImageGrid();
        const imageGrid = document.getElementById('imageGrid');

        // Check that drag and drop area exists
        expect(imageGrid.querySelector('.drag-drop-area')).toBeInTheDocument();

        // Check each image in imageData
        imageData.forEach(image => {
            const imgElement = imageGrid.querySelector(`img[alt="${image.name}"]`);
            expect(imgElement).toBeInTheDocument();
            expect(imgElement.getAttribute('src')).toBe(image.path);
        });
    });
});

// Test for initializeAlgorithmStatus
describe('initializeAlgorithmStatus', () => {
    test('renders algorithm statuses correctly', () => {
        initializeAlgorithmStatus();
        const algorithmStatus = document.getElementById('algorithmStatus');

        // Check each algorithm in dashboardData
        dashboardData.algorithms.forEach(algo => {
            expect(algorithmStatus).toHaveTextContent(algo.name);
            expect(algorithmStatus).toHaveTextContent(algo.status);
            if (algo.lastRun) expect(algorithmStatus).toHaveTextContent(algo.lastRun);
            if (algo.accuracy) expect(algorithmStatus).toHaveTextContent(algo.accuracy);
        });
    });
});

// Test for initializeRecentComparisons
describe('initializeRecentComparisons', () => {
    test('renders recent comparisons correctly', () => {
        initializeRecentComparisons();
        const recentComparisons = document.getElementById('recentComparisons');

        // Check each comparison in dashboardData
        dashboardData.recentComparisons.forEach(comp => {
            expect(recentComparisons).toHaveTextContent(comp.user);
            expect(recentComparisons).toHaveTextContent(comp.time);
        });
    });
});
