@router.post("/tickets")
def createTicket(ticket: TicketCreate, db: Session = Depends(get_db), user: User = Depends(getCurrentUser)):
    user = db.query(User).filter(User.id == ticket.uid).first()
    if not user:
        raise HTTPException(status_code=404, detail="User not found. Could not create a ticket")


    dbTicket = Ticket(uid=ticket.uid, priority=ticket.priority, customerName = ticket.customerName, notes = ticket.notes, issueDate = datetime.now(), status = "Draft")
    db.add(dbTicket)
    db.commit()
    db.refresh(dbTicket)
    
     # Log activity
    new_activity = TicketActivity(
        ticket_id=dbTicket.id,
        user_id=user.id,
        activity_type=ActivityType.STATUS_CHANGED_SYSTEM.value,
        description="Ticket's status has been marked as Draft by system."
    )
     # Log activity
    new_activity_status = TicketActivity(
        ticket_id=dbTicket.id,
        user_id=user.id,
        activity_type=ActivityType.CREATED.value,
        description=f"Ticket created for customer {ticket.customerName}"
    )
    db.add(new_activity_status)
    db.add(new_activity)
        
    notification_message = f"Ticket has been created and status has been marked as Draft by system for {ticket.customerName}"
    notification = Notification(
        user_id=ticket.uid,
        message=notification_message,
        link=f"/ticket/{dbTicket.id}",
        type="status_change",
        ticket_id=dbTicket.id,
        created_at=datetime.utcnow(),
        is_read=False
        )
    db.add(notification)
    
    send_notification_email(db, ticket.uid, notification_message, dbTicket.id, "status_change")  
    
    if user.managerId:
        notification = Notification(
            user_id=user.managerId,
            message=notification_message,
            link=f"/ticket/{dbTicket.id}",
            type="comment",
            ticket_id=dbTicket.id,
            created_at=datetime.utcnow(),
            is_read=False
            )
        db.add(notification)
        
        send_notification_email(db, user.managerId, notification_message, dbTicket.id, "comment")
    
    db.commit()
    
    return {"id": dbTicket.id, "message": "Ticket created successfully"}


@router.get("/tickets/{ticket_id}")
def get_ticket_details(ticket_id: int, db: Session = Depends(get_db)):
    ticket = db.query(Ticket).filter(Ticket.id == ticket_id).first()


    if ticket:
        user = db.query(User).filter(User.id == ticket.uid).first()
        if user:
            ticket.firstName = user.firstName
            ticket.lastName = user.lastName
        return ticket
    else:
        raise HTTPException(status_code=404, detail="Ticket not found")
