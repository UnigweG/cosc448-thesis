class PDFComparisonService:
    def __init__(self, similarity_threshold=0.75):
        self.similarity_threshold = similarity_threshold
        self.temp_dir = os.path.join(os.getcwd(), 'temp')
        os.makedirs(self.temp_dir, exist_ok=True)

        # Insurance-related keywords for relevance filtering
        self.relevant_keywords = [
            "premium", "premiums",
            "deductible", "deductibles",
            "limit", "limits",
            "coverage", "coverages",
            "liability", "policy", "exclusion",
            "endorsement", "loss", "property"
        ]

        # Regex patterns for special content
        self.money_regex = re.compile(r'(\$?\d{1,3}(,\d{3})*(\.\d{2})?)')
        self.date_regex = re.compile(r'(\b\d{1,2}/\d{1,2}/\d{2,4}\b)|(\b\d{4}-\d{1,2}-\d{1,2}\b)')
        self.date_period_regex = re.compile(r'(\b\d{1,2}/\d{1,2}/\d{2,4}\b\s*-\s*\b\d{1,2}/\d{1,2}/\d{2,4}\b)')

    def normalize_text(self, text):
        """Normalize text by handling whitespace and common formatting issues."""
        text = re.sub(r'\s+', ' ', text)
        text = re.sub(r'(\w+)\s+\'\s*s\b', r"\1's", text)
        text = re.sub(r'\s+([.,;:])', r'\1', text)
        return text.strip()

    def calculate_similarity(self, sentence1, sentence2):
        """Calculate similarity between two sentences using Levenshtein distance."""
        sent1 = self.normalize_text(sentence1)
        sent2 = self.normalize_text(sentence2)
        max_len = max(len(sent1), len(sent2))
        if max_len == 0:
            return 0.0

        distance = Levenshtein.distance(sent1, sent2)
        similarity = 1 - (distance / max_len)
        return similarity

    def find_similar_sentences(self, old_sentences, new_sentences):
        """Find similar sentences between old and new text using Levenshtein distance."""
        matches = []
        unmatched_old = list(old_sentences)
        unmatched_new = list(new_sentences)

        for old_sent in old_sentences:
            best_match = None
            best_similarity = 0
            best_match_index = -1

            for j, new_sent in enumerate(unmatched_new):
                sim = self.calculate_similarity(old_sent, new_sent)
                if sim > best_similarity and sim >= self.similarity_threshold:
                    best_match = new_sent
                    best_similarity = sim
                    best_match_index = j

            if best_match:
                matches.append({
                    'old_sentence': old_sent,
                    'new_sentence': best_match,
                    'similarity': best_similarity
                })
                if best_match_index != -1:
                    del unmatched_new[best_match_index]
                unmatched_old.remove(old_sent)

        return matches, unmatched_old, unmatched_new
