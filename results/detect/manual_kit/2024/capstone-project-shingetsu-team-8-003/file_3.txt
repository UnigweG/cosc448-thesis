# Tests for the demographic route
def test_demographic_route(client, init_database):
    """Test the GET route for demographic data."""

    # Delete already existing data
    DemographicData.query.delete()
    db.session.commit()

    # Add mock data to the database
    demographic = DemographicData(
        fname="John", lname="Doe", age=30, country="US", gender="Male", uniCourse="Computer Science"
    )
    db.session.add(demographic)
    db.session.commit()

    # Make a GET request to fetch demographic data
    response = client.get('/api/demographic')
    assert response.status_code == 200

    # Verify the response data
    data = response.get_json()
    assert isinstance(data, list)  # Ensure the response is a list
    assert len(data) == 1  # Ensure one record is returned
    assert data[0]["fname"] == "John"
    assert data[0]["lname"] == "Doe"
    assert data[0]["age"] == '30'
    assert data[0]["country"] == "US"
    assert data[0]["gender"] == "Male"
    assert data[0]["uniCourse"] == "Computer Science"
