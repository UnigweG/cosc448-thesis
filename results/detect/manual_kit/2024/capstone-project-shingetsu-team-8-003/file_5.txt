class TestStatisticalAnalysis(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        """Set up test environment once for all tests"""
        cls.app = create_app()
        cls.app_context = cls.app.app_context()
        cls.app_context.push()

    @classmethod
    def tearDownClass(cls):
        """Clean up after all tests"""
        cls.app_context.pop()

    def setUp(self):
        """Set up before each test"""
        self.client = self.app.test_client()
        db.create_all()
        self._populate_test_data()

    def tearDown(self):
        """Clean up after each test"""
        db.session.remove()
        db.drop_all()

    def _create_test_questions(self):
        """Create test questions with controlled content"""
        questions = [
            ImageQuestion(question_text="Which image do you find more aesthetically pleasing?"),
            ImageQuestion(question_text="Select the image that you prefer."),
            ImageQuestion(question_text="Choose the image that looks more balanced."),
            ImageQuestion(question_text="Which image appears more visually appealing?"),
            ImageQuestion(question_text="Select the most aesthetically pleasing image.")  # Similar to first question
        ]
        for q in questions:
            db.session.add(q)
        db.session.commit()
        return questions

    def _create_test_images(self, questions):
        """Create test images with controlled distribution"""
        images = []
        for q in questions:
            # Create two images per question
            for i in range(2):
                img = Image(
                    url=f"test_image_{q.id}_{i+1}.jpg",
                    question_id=q.id
                )
                images.append(img)
                db.session.add(img)
        db.session.commit()
        return images
