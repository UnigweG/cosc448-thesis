def validate_question_id(question_id):
    """
    Validate if a question ID exists in the database
    Args:
        question_id: ID to validate
    Returns:
        tuple: (bool, str) - (is_valid, error_message)
    """
    question = ImageQuestion.query.get(question_id)
    if not question:
        return False, f"Question with ID {question_id} not found"
    return True, None

def validate_user_id(user_id):
    """
    Validate if a user ID exists in the demographic data
    Args:
        user_id: ID to validate
    Returns:
        tuple: (bool, str) - (is_valid, error_message)
    """
    user = DemographicData.query.get(user_id)
    if not user:
        return False, f"User with ID {user_id} not found"
    return True, None

def validate_image_id(image_id):
    """
    Validate if an image ID exists in the database
    Args:
        image_id: ID to validate
    Returns:
        tuple: (bool, str) - (is_valid, error_message)
    """
    image = Image.query.get(image_id)
    if not image:
        return False, f"Image with ID {image_id} not found"
    return True, None

def validate_response_data(response):
    """
    Validate all IDs in a user response
    Args:
        response: UserResponse object
    Returns:
        tuple: (bool, str) - (is_valid, error_message)
    """
    # Validate question ID
    valid, error = validate_question_id(response.question_id)
    if not valid:
        return False, error
        
    # Validate user ID
    valid, error = validate_user_id(response.user_id)
    if not valid:
        return False, error
        
    # Validate image ID
    valid, error = validate_image_id(response.image_id)
    if not valid:
        return False, error
        
    return True, None

def calculate_entropy(responses):
    """
    Calculate Shannon entropy for a distribution of responses
    Args:
        responses: list of response counts for each answer choice
    Returns:
        entropy value
    """
    total = sum(responses)
    if total == 0:
        return 0
    
    probabilities = [r / total for r in responses]
    entropy = -sum(p * math.log2(p) if p > 0 else 0 for p in probabilities)
    return entropy

def cramers_v(confusion_matrix):
    """
    Calculate Cramer's V statistic for categorical correlation
    Args:
        confusion_matrix: contingency table of two questions' responses
    Returns:
        Cramer's V value between 0 and 1
    """
    chi2 = chi2_contingency(confusion_matrix)[0]
    n = confusion_matrix.sum()
    min_dim = min(confusion_matrix.shape) - 1
    if n == 0 or min_dim == 0:
        return 0
    return np.sqrt(chi2 / (n * min_dim))
