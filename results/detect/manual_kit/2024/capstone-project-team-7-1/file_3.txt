// Extended Transaction interface that includes account_type property
interface ExtendedTransaction extends Transaction {
  account_type?: string;
}

export function TransactionSummary() {
    // Get data and functions from context instead of props
    const {
        transactions: allTransactions,
        loading, 
        loadingMore,
        error,
        loadMoreTransactions,
        formatDate,
        formatAmount,
        getCategoryIcon,
        getAmountColor
    } = useTransactions();
    
    // Get account data to map account_id to account names
    const {
        accountData,
        formatAccountType
    } = useAccounts();
    
    // Create a map of account_id to account name and type
    const accountMap = useMemo(() => {
        const map = new Map<string, { name: string, type: string }>();
        if (accountData) {
            accountData.forEach(account => {
                map.set(account.account_id, { 
                    name: account.name || account.mask || `Account ${account.account_id.substring(0, 4)}`, 
                    type: account.type || "Unknown"
                });
            });
        }
        return map;
    }, [accountData]);
    
    // Enhance transactions with account information
    const enhancedTransactions = useMemo(() => {
        if (!allTransactions) return null;
        
        return allTransactions.map(transaction => {
            const accountInfo = accountMap.get(transaction.account_id);
            return {
                ...transaction,
                account_name: accountInfo?.name || transaction.account_name || `Account ${transaction.account_id.substring(0, 4)}`,
                account_type: accountInfo?.type ? formatAccountType(accountInfo.type) : undefined
            } as ExtendedTransaction;
        });
    }, [allTransactions, accountMap, formatAccountType]);

    // State to track how many transactions to display
    const [displayCount, setDisplayCount] = useState(8);

    // State for download dropdown
    const [isDownloadOpen, setIsDownloadOpen] = useState(false);
    const [dateRange, setDateRange] = useState("last30days");
    const [downloadFormat, setDownloadFormat] = useState("csv");

    // State for table view dialog
    const [isTableViewOpen, setIsTableViewOpen] = useState(false);
    
    // State for table filters
    const [isFiltersOpen, setIsFiltersOpen] = useState(false);
    const [filterCount, setFilterCount] = useState(0);
    
    // Sorting state
    const [sortField, setSortField] = useState<string>("date");
    const [sortDirection, setSortDirection] = useState<"asc" | "desc">("desc");

    // Filter states
    const [categoryFilters, setCategoryFilters] = useState<{[key: string]: boolean}>({});
