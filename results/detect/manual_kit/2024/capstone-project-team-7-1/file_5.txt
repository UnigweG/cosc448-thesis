function processStreamChunks(chunks: string): { [key: string]: any } {
  const result: { [key: string]: any } = {};
  const chunkArray = chunks.split('}{').map(chunk => chunk.replace(/^\{|\}$/g, ''));
  for (const chunk of chunkArray) {
    try {
      const parsedChunk = JSON.parse(`{${chunk}}`);
      const { type, content } = parsedChunk;
      if (result[type]) {
        result[type] += content;
      } else {
        result[type] = content;
      }
    } catch (error) {
      console.error("Error parsing chunk:", chunk, error);
    }
  }
  return result;
}
