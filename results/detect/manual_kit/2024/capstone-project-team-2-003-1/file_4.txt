'use client';

import React, { useEffect, useState } from 'react';
import { Card } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import {
  Home, Utensils, Car, Gamepad, Plus, AlertCircle,
  PieChart, Download, Edit2, Trash2, Wallet, Check
} from 'lucide-react';
import { useAuth } from '@/components/lib/AuthContext';
import Papa from 'papaparse';

interface Budget {
  id: number;
  name: string;
  current_amount: number;
  max_amount: number;
  icon: string;
  color: string;
}

export default function BudgetPage() {
  const { user } = useAuth();
  const [budgets, setBudgets] = useState<Budget[]>([]);
  const [loading, setLoading] = useState(true);
  const [showAddForm, setShowAddForm] = useState(false);
  const [showEditForm, setShowEditForm] = useState(false);
  const [selectedBudget, setSelectedBudget] = useState<Budget | null>(null);
  const [totalBudgeted, setTotalBudgeted] = useState(0);
  const [totalSpent, setTotalSpent] = useState(0);
  
  const [newBudget, setNewBudget] = useState({
    name: '',
    max_amount: '',
    icon: 'home',
    color: 'blue',
  });

  const [editAmount, setEditAmount] = useState('');

  const userid = user?.id ?? '';

  useEffect(() => {
    if (userid) {
      fetchBudgets();
    }
  }, [userid]);

  useEffect(() => {
    if (budgets.length > 0) {
      const total = budgets.reduce((sum, budget) => sum + budget.max_amount, 0);
      const spent = budgets.reduce((sum, budget) => sum + budget.current_amount, 0);
      setTotalBudgeted(total);
      setTotalSpent(spent);
    }
  }, [budgets]);

  const fetchBudgets = async () => {
    try {
      setLoading(true);
      const response = await fetch('/api/budget', {
        headers: {
          'x-user-id': userid
        }
      });
      if (!response.ok) throw new Error('Failed to fetch budgets');
      const data = await response.json();
      setBudgets(data.response);
    } catch (error) {
      console.error('Failed to fetch budgets:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleCreateBudget = async (e: React.FormEvent) => {
    e.preventDefault();
    try {
      const response = await fetch('/api/budget', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'x-user-id': userid
        },
        body: JSON.stringify({
          ...newBudget,
          max_amount: parseFloat(newBudget.max_amount),
        }),
      });

      if (!response.ok) throw new Error('Failed to create budget');
      
      await fetchBudgets();
      setShowAddForm(false);
      setNewBudget({ name: '', max_amount: '', icon: 'home', color: 'blue' });
    } catch (error) {
      console.error('Failed to create budget:', error);
    }
  };

  const handleUpdateAmount = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!selectedBudget) return;

    try {
      const response = await fetch('/api/budget/update-amount', {
        method: 'PUT',
        headers: {
          'Content-Type': 'application/json',
          'x-user-id': userid
        },
        body: JSON.stringify({
          id: selectedBudget.id,
          current_amount: parseFloat(editAmount)
        }),
      });

      if (!response.ok) throw new Error('Failed to update amount');
      
      await fetchBudgets();
      setShowEditForm(false);
      setSelectedBudget(null);
      setEditAmount('');
    } catch (error) {
      console.error('Failed to update amount:', error);
    }
  };

  const handleDeleteBudget = async (budgetId: number) => {
    if (!confirm('Are you sure you want to delete this budget?')) return;
    
    try {
      const response = await fetch('/api/budget', {
        method: 'DELETE',
        headers: {
          'Content-Type': 'application/json',
          'x-user-id': userid
        },
        body: JSON.stringify({ id: budgetId }),
      });

      if (!response.ok) throw new Error('Failed to delete budget');
      await fetchBudgets();
    } catch (error) {
      console.error('Failed to delete budget:', error);
    }
  };

  const exportReport = () => {
    const reportData = budgets.map(budget => ({
      Category: budget.name,
      'Current Amount': budget.current_amount,
      'Maximum Amount': budget.max_amount,
      'Remaining Amount': budget.max_amount - budget.current_amount,
      'Usage Percentage': ((budget.current_amount / budget.max_amount) * 100).toFixed(2) + '%'
    }));

    const csv = Papa.unparse(reportData);
