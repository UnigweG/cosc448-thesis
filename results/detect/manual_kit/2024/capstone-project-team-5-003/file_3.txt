function Survey_dashboard() {
  const location = useLocation();
  const navigate = useNavigate();
  const { language } = useLanguage(); // Get current language
  const queryParams = new URLSearchParams(location.search);
  const initialTab = queryParams.get('tab') || 'summary';

  const [activeTab, setActiveTab] = useState(initialTab);
  const [isDark, setIsDark] = useState(() => localStorage.getItem("isDark") === "true");
  const { surveyId } = useParams();
  console.log('Extracted surveyId:', surveyId);
  const handleTabClick = (tabName) => {
    setActiveTab(tabName);
  };

  useEffect(() => {
    setActiveTab(initialTab);
  }, [location.search]);

  const handleTabChange = (tab) => {
    setActiveTab(tab);
    navigate(`/surveyResults/${surveyId}?tab=${tab}`);
  };

  // Toggle dark mode based on `isDark`
  useEffect(() => {
    document.body.setAttribute("data-theme", isDark ? "dark" : "light");
    localStorage.setItem("isDark", isDark); // Ensure `localStorage` is updated
  }, [isDark]);

  return (
    <div className="resultDashboard">
      <div className="tab-container">
        <div className="tab-header">
          <div className = "title">
            <h1>{translations[language].dashboard}</h1>
          </div>
          <div className = "selectionContainer">
          <div className="selection">
            <button
              className={activeTab === 'summary' ? 'active' : ''}
              onClick={() => handleTabChange('summary')}
            >
              {translations[language].overview}
            </button>
            <button
              className={activeTab === 'results' ? 'active' : ''}
              onClick={() => handleTabChange('results')}
            >
              {translations[language].surveyResults}
            </button>
            <button
              className={activeTab === 'question' ? 'active' : ''}
              onClick={() => handleTabChange('question')}
            >
              Question
            </button>
            <button
              className={activeTab === 'report' ? 'active' : ''}
              onClick={() => handleTabChange('report')}
            >
              {translations[language].report}
            </button>
          </div>
            
          </div>
        </div>

        {/* Tab Content */}
        <div className='tab-content ${activeTab}'>
          {activeTab === 'summary' && <SummaryQuestions surveyId={surveyId} />}
          {activeTab === 'results' && <SurveyResults />}
          {activeTab === 'report' && <Report />}
          {activeTab === 'question' && <Question />}
        </div>
      </div>
    </div>
  );
}
