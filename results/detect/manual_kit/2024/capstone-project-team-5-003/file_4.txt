const SurveyResultsAllData = () => {

  const [boxPlotData, setBoxPlotData] = useState([]);

  useEffect(() => {
    const fetchHighestVarianceData = async () => {
      try {
        const response = await fetch(`http://localhost:5040/top-questions/${surveyId}`);
        if (!response.ok) {
          throw new Error('Network response was not ok');
        }
        const data = await response.json();
        setTopQuestion(data);  // Set the data to state
        const firstQuestionId = data.length > 0 ? data[0].question_id : 0; // Default to 0 if no valid ID
        setHighestVarianceQuestion(firstQuestionId); // Ensure it's always a number
      } catch (error) {
        setError(error.message); // Handle errors
      } finally {
        setLoading(false); // Stop loading after data is fetched
      }
    };
    const fetchLowestVarianceData = async () => {
      try {
        const response = await fetch(`http://localhost:5040/lowest-variance-questions`);
        if (!response.ok) {
          throw new Error('Network response was not good. ');
        }
        const data = await response.json();
        setTopQuestionLV(data);
        const firstQuestionId = data.length > 0 ? data[0].question_id : 0;
        setLowestVarianceQuestion(firstQuestionId);
      } catch (error) {
        setError(error.message);
      } finally {
        setLoading(false);
      }
    };
    const fetchFastestData = async () => {
      try {
        const response = await fetch(`http://localhost:5040/fastestQuestion/${surveyId}`);
        if (!response.ok) {
          throw new Error('Network response was not good');
        }
        const data = await response.json();
        const formattedData = data.map((item) => ({
          question_id: `Q${item.question_id}`,
          avg_response_time: item.avg_response_time / 1000, // Convert to milliseconds if needed
        }));
        setFastestQuestionsChartData(formattedData);
      } catch (error) {
        setError(error.message);
      } finally {
        setLoading(false);
      }
    };

        const response = await fetch(`http://localhost:5040/slowestQuestion/${surveyId}`);

          throw new Error('Network response was not good');

          question_id: `Q${item.question_id}`,
          avg_response_time: item.avg_response_time / 1000, // Convert to milliseconds if needed
        }));
        setSlowestQuestionsChartData(formattedData);
      } catch (error) {
        setError(error.message);
      } finally {
        setLoading(false);
      }
    };

    fetchHighestVarianceData();
