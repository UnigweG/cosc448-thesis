# the generator model creates a 128x128 RBG image with the noise vector
# each layer is supposed to learn how to fill in details during upsampling.
def makeGenerator():
    model = tf.keras.Sequential([
        # create a stack of layers where each layers output is used by tyhe next layer
        layers.Dense(8 * 8 * 1024, use_bias=False, input_shape=(NOISE_DIM,)),
        layers.BatchNormalization(),
        layers.ReLU(),
        layers.Reshape((8, 8, 1024)),

        layers.Conv2DTranspose(512, (5, 5), strides=(2, 2), padding='same', use_bias=False),
        layers.BatchNormalization(),# batch normalizaition to stabilize training
        layers.ReLU(),# added for nonlinearity

        layers.Conv2DTranspose(256, (5, 5), strides=(2, 2), padding='same', use_bias=False),
        layers.BatchNormalization(),# batch normalizaition to stabilize training
        layers.ReLU(),# added for nonlinearity

        layers.Conv2DTranspose(128, (5, 5), strides=(2, 2), padding='same', use_bias=False),
        layers.BatchNormalization(),
        layers.ReLU(),

        layers.Conv2DTranspose(3, (5, 5), strides=(2, 2), padding='same', use_bias=False, activation='tanh')
    ])
    return model

# this will blur the inputs to the discriminator making it 'less confident'
# supposed to help mode collapse
class noise_for_discriminator(tf.keras.Layer): # make custome layer
  def __init__(self, stddev=0.1): # define the noise and how strong it is
    super().__init__() # call constructor
    self.noise=tf.keras.layers.GaussianNoise(stddev) # built in gaussiannoise layer

  def call(self,x,training=False):
    return self.noise(x,training=training) # only add noise when it train

# discriminator model takes a 128x128 RGB image determines if it is fake or real by outputting 0 or 1
def makeDiscriminator():
    model = tf.keras.Sequential([
        noise_for_discriminator(0.1),
        layers.Conv2D(64, (5, 5), strides=(2, 2), padding='same', input_shape=(128, 128, 3)),
        layers.LeakyReLU(alpha=0.2),
        layers.Dropout(0.3),

        layers.Conv2D(128, (5, 5), strides=(2, 2), padding='same'),
        layers.LeakyReLU(alpha=0.2),
        layers.Dropout(0.3),

        layers.Conv2D(256, (5, 5), strides=(2, 2), padding='same'),
        layers.LeakyReLU(alpha=0.2),
        layers.Dropout(0.3),

        layers.Conv2D(512, (5, 5), strides=(2, 2), padding='same'),
        layers.LeakyReLU(alpha=0.2),
        layers.Dropout(0.3),

        layers.Flatten(),
        layers.Dense(1)
    ])
    return model

# loss functions

# create cross enthrophy loss used for classification as real vs fake
CE = tf.keras.losses.BinaryCrossentropy(from_logits=True)

# I have been trying to tone down the disciminator for a while by trying to use practices to make it more
# confused. With more or lesss success lol
def discriminator_loss(RO, FO):
    RL = tf.ones_like(RO)*0.9 # smooth the labels a little to make discriminator more confused
    FL = tf.zeros_like(FO) # the fake label stays 0

    flip = tf.random.uniform(shape=RL.shape)<0.05 # flip randomly 5% of the labels
    RL = tf.where(flip, tf.zeros_like(RL), RL)
    return CE(RL, RO) + CE(FL, FO)

def generator_loss(FO):
    return CE(tf.ones_like(FO), FO)
