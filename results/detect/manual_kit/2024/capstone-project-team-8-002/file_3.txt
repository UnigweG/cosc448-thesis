# the generator model creates a 128x128 RBG image with the noise vector
# each layer is supposed to learn how to fill in details during upsampling.
def makeGenerator():
    model = tf.keras.Sequential([
        # create a stack of layers where each layers output is used by tyhe next layer
        layers.Dense(8 * 8 * 1024, use_bias=False, input_shape=(NOISE_DIM,)), # fully connected layer projected into a 3d tensor
        layers.BatchNormalization(), # normalize activations during training to help stabilize
        layers.ReLU(), # adds non linearity so the model can learn complex mappings
        layers.Reshape((8, 8, 1024)), # reshape the dense layer into the feature map

        # upsampling layer to grow the image to high resolution, with kernel size 5,5, double spatial dimensions with strides 2,2
        layers.Conv2DTranspose(512, (5, 5), strides=(2, 2), padding='same', use_bias=False),
        layers.BatchNormalization(),
        layers.ReLU(),

        layers.Conv2DTranspose(256, (5, 5), strides=(2, 2), padding='same', use_bias=False),
        layers.BatchNormalization(),
        layers.ReLU(),

        # decrease the channels from 512 to 256 to 128 to work on finer details
        layers.Conv2DTranspose(128, (5, 5), strides=(2, 2), padding='same', use_bias=False),
        layers.BatchNormalization(),
        layers.ReLU(),

        layers.Conv2DTranspose(64, (5, 5), strides=(2, 2), padding='same', use_bias=False),
        layers.BatchNormalization(),
        layers.ReLU(),

        # final upsamplin with output matching the input data format [-1,1]
        layers.Conv2DTranspose(3, (5, 5), strides=(2, 2), padding='same', use_bias=False, activation='tanh')
    ])
    return model

# this will blur the inputs to the discriminator making it 'less confident'
# supposed to help against mode collapse I have been struggling with
class noise_for_discriminator(tf.keras.Layer): # make custom layer
  def __init__(self, stddev=0.1): # define the noise and how strong it is
    super().__init__() # call constructor
    self.noise=tf.keras.layers.GaussianNoise(stddev) # built in gaussiannoise layer

  def call(self,x,training=False):
    return self.noise(x,training=training) # only add noise when it train

# discriminator model takes a 128x128 RGB image (real or fake) determines if it is fake or real
# the discriminator should be hard to fool which will help the generator with generating better images
