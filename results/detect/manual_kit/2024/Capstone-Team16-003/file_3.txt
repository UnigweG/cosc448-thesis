const ComparePage = () => {
    const {authState} = useAuth();
    const {documentList, setDocumentList, refreshDocumentList} = useDocuments();
    const [filteredDocuments, setFilteredDocuments] = useState([]);
    const [sourceDoc, setSourceDoc] = useState(null);
    const [targetDoc, setTargetDoc] = useState(null);
    const [searchParams, setSearchParams] = useSearchParams();
    const {t} = useTranslation();
    const {setHeaderContent} = useHeaderContent();
    const navigate = useNavigate();
    const {onEvent} = useEventContext();

    const [aiOverview, setAiOverview] = useState({});
    const [isAiPanelOpen, setIsAiPanelOpen] = useState(false);
    const [loadingAi, setLoadingAi] = useState(false);
    const [currentComparisonId, setCurrentComparisonId] = useState(null);

    const [manualComparisons, setManualComparisons] = useState([]);

    const customer = searchParams.get('customer');
    const source = searchParams.get('source');
    const target = searchParams.get('target');

    useEffect(() => {
        async function fetchData() {
            const sourceDoc = await getDocument(source);
            if (!sourceDoc.success) {
                console.error('Failed to retrieve source document');
                setSourceDoc({absent: true})
                return;
            }
            setSourceDoc({
                name: sourceDoc.document.getName(),
                file: sourceDoc.document.getFile(),
                parsed: sourceDoc.document.getParsed(),
                id: sourceDoc.document.getId()
            });

            if (target) {
                const targetDoc = await getDocument(target);
                if (!targetDoc.success) {
                    console.error('Failed to retrieve target document');
                    setTargetDoc({absent: true})
                    return;
                }
                setTargetDoc({
                    name: targetDoc.document.getName(),
                    file: targetDoc.document.getFile(),
                    parsed: targetDoc.document.getParsed(),
                    id: targetDoc.document.getId()
                });

                const comparisonData = await getComparisonData(source, target);
                if (!comparisonData.success) {
                    console.error('Failed to retrieve comparison data', comparisonData.message);
                    return;
                }
                const comparisons = Comparison.fromJsonArray(comparisonData.data);
                setManualComparisons(Comparison.getManualComparisons(comparisons));
            } else {
                setTargetDoc({absent: true});
            }
        }

        fetchData().then();

    }, [searchParams]);

    useEffect(() => {
        setHeaderContent(<div className="compare-documents-title">
            {t("compare_page.title")}
        </div>);

        return () => setHeaderContent(null);
    }, [setHeaderContent]);

    const updateFilteredDocuments = () => {
        const index = documentList.findIndex(doc => `${doc.getId()}` === source);
        if (index === -1) return;

        const filtered = documentList.slice();
        filtered.splice(index, 1);
        setFilteredDocuments(filtered);
    }

    useEffect(() => {
        if (documentList) {
            updateFilteredDocuments();
        }
