    public class KSDbContext : IdentityDbContext<User, IdentityRole, string>, IKSDbContext
    {
        public DbSet<User> Users { get; set; }
        public DbSet<RefreshToken> RefreshTokens { get; set; }

        public KSDbContext(
            DbContextOptions<KSDbContext> options
            ) : base(options)
        {
            
        }

        public async Task SeedAsync(
            RoleManager<IdentityRole> roleManager,
            UserManager<User> userManager,
            CancellationToken cancellationToken)
        {
            Console.WriteLine("Starting database seeding...");

            // 1. Check and create roles
            var roles = new[] { "Admin", "Vendor", "Vendor Manager", "Vendor Under Manager" };

            foreach (var roleName in roles)
            {
                if (!await roleManager.RoleExistsAsync(roleName))
                {
                    Console.WriteLine($"Role '{roleName}' does not exist. Creating '{roleName}' role...");
                    var roleResult = await roleManager.CreateAsync(new IdentityRole(roleName));
                    if (roleResult.Succeeded)
                    {
                        Console.WriteLine($"Role '{roleName}' created successfully.");
                    }
                    else
                    {
                        Console.WriteLine($"Failed to create role '{roleName}'. Errors:");
                        foreach (var error in roleResult.Errors)
                        {
                            Console.WriteLine($"- {error.Description}");
                        }
                    }
                }
                else
                {
                    Console.WriteLine($"Role '{roleName}' already exists.");
                }
            }

            // 2. Seed Admin User
            var adminEmail = "admin@admin.com";
            var adminUser = await userManager.FindByEmailAsync(adminEmail);
            if (adminUser == null)
            {
                Console.WriteLine("Admin user does not exist. Creating admin user...");
                adminUser = new User
                {
                    UserName = adminEmail,
                    Email = adminEmail,
                    EmailConfirmed = true,
                    DateCreated = DateTime.Now,
                };

                var createAdminResult = await userManager.CreateAsync(adminUser, "Testing123!");
                if (createAdminResult.Succeeded)
                {
                    Console.WriteLine("Admin user created successfully.");

                    // Assigning Admin role
                    Console.WriteLine("Assigning Admin role...");
                    var addToRoleResult = await userManager.AddToRoleAsync(adminUser, "Admin");
