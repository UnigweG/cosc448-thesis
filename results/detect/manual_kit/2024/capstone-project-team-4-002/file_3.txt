def get_centered_grid_positions(num_shapes, canvas_size):
    rows = math.ceil(math.sqrt(num_shapes))
    spacing_y = canvas_size / (rows + 1)
    positions = []
    remaining = num_shapes
    for i in range(rows):
        per_row = math.ceil(num_shapes / rows)
        shapes_in_row = per_row if remaining >= per_row else remaining
        spacing_x = canvas_size / (shapes_in_row + 1)
        for j in range(shapes_in_row):
            x = int((j + 1) * spacing_x)
            y = int((i + 1) * spacing_y)
            positions.append((x, y))
        remaining -= shapes_in_row
    return positions

def add_complexity(points, comp_factor):
    if comp_factor <= 0:
        return points
    new_points = []
    n = len(points)
    for i in range(n):
        p1 = points[i]
        p2 = points[(i+1) % n]
        new_points.append(p1)
        mid_x = (p1[0] + p2[0]) / 2
        mid_y = (p1[1] + p2[1]) / 2
        edge_length = math.hypot(p2[0]-p1[0], p2[1]-p1[1])
        max_disp = edge_length * 0.3 * comp_factor
        disp_x = random.uniform(-max_disp, max_disp)
        disp_y = random.uniform(-max_disp, max_disp)
        new_points.append((mid_x + disp_x, mid_y + disp_y))
    return new_points

def inset_polygon(points, center, scale):
    cx, cy = center
    return [(cx + (x - cx) * scale, cy + (y - cy) * scale) for (x, y) in points]

def inset_bbox(bbox, center, scale):
    cx, cy = center
    x0, y0, x1, y1 = bbox
    return (cx - (cx - x0) * scale,
            cy - (cy - y0) * scale,
            cx + (x1 - cx) * scale,
            cy + (y1 - cy) * scale)
