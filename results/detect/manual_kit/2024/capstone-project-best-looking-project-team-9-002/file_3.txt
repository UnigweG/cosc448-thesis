/** 
 * A custom tooltip component that displays an image preview 
 * and the vote count when hovering over a slice/bar.
 */
const ImageTooltip = ({ active, payload }) => {
  if (active && payload && payload.length) {
    const { imageUrl, value, selectedByUser } = payload[0].payload;
    return (
      <Box bg="white" p={2} borderRadius="md" boxShadow="md">
        <Image src={imageUrl} alt="Option" boxSize="80px" mb={2} />
        <Text color="black">{value} votes</Text>
        {selectedByUser && (
          <Text color="green.500" fontWeight="bold">
            Your Choice
          </Text>
        )}
      </Box>
    );
  }
  return null;
};

    try {
