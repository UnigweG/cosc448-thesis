describe("Add Item to Plaid", () => {
  it("invalid userid", async () => {
    const user = await createTestUser();
    const res = await addItem("testitemid", user.id + 100, "testtoken");
    expect(res.error).toEqual(`User with id ${user.id + 100} not found.`);
  });

  it("add item successfully", async () => {
    const user = await createTestUser();
    const userId = user.id;
    const res = await addItem("testitemid", userId, "testtoken");
    const dbItem = await prisma.Item.findUnique({
      where: {
        id: "testitemid",
      },
    });
    expect(dbItem.accessToken).toEqual("testtoken");
  });
});

describe("Remove item", () => {
  it("invalid itemid", async () => {
    const user = await createTestUser();
    const res = await removeItem("wrongitemid");
    expect(res[1].error).toEqual("Item with id wrongitemid not found.");
    expect(res[0].error).toEqual("Item not found in Plaid.");
  });

  it("remove item successfully", async () => {
    const user = await createTestUser();
    const loginResponse = await agent.post("/users/login").send({
      email: user.email,
      password: user.password,
    });
    const authToken = loginResponse.body.token;
    // create item in Plaid
    const token = await getTestPublicToken();
    await agent
      .post("/plaidtokens/exchange_public_token")
      .send({
        publicToken: token.public_token,
      })
      .set("Cookie", [`transactionQueryingToken=${authToken}`]);
    const items = await getItemsForUser(user.id);
    const item = items[0];
    await removeItem(item.accessToken);
    const dbItem = await prisma.Item.findUnique({
      where: {
        id: item.id,
      },
    });
    // try to get item from plaid
    const response = await getItemFromPlaid(item.accessToken);
    await clearItems();
    expect(dbItem).toEqual(null);
    expect(response.error_code).toEqual("ITEM_NOT_FOUND");
  });
});
