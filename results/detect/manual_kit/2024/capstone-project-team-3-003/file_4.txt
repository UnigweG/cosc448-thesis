    //  .set("Cookie", [`transactionQueryingToken=${token}`]);
    //expect(response.statusCode).toBe(422);
    //expect(response.body.error).toBe("Email not in database.");

    const response1 = await agent.delete("/users").send({
      email: user.email,
      password: "bogus",
    })
      .set("Cookie", [`transactionQueryingToken=${token}`]);
    expect(response1.statusCode).toBe(401);
    expect(response1.body.error).toBe("Incorrect password.");
  });

  it("Delete account", async () => {
    const user = await createTestUser();
    const res = await agent.post("/users/login").send({
      email: user.email,
      password: user.password,
    });
    const token = res.body.token;
    const response = await agent.delete("/users").send({
      email: user.email,
      password: user.password,
    })
      .set("Cookie", [`transactionQueryingToken=${token}`]);
    expect(response.statusCode).toBe(200);
    expect(response.text).toBe("Account deleted");
  });

  it("Delete account with plaid items", async () => {
    const user = await createTestUser();
    const loginResponse = await agent.post("/users/login").send({
      email: user.email,
      password: user.password,
    });
    const authToken = loginResponse.body.token;
    // create item in Plaid
    const token = await getTestPublicToken();
    await agent
      .post("/plaidtokens/exchange_public_token")
      .send({
        publicToken: token.public_token,
      })
      .set("Cookie", [`transactionQueryingToken=${authToken}`]);
    const items = await getItemsForUser(user.id);
    const item = items[0];
    await agent.delete("/users").send({
      email: user.email,
      password: user.password,
    })
      .set("Cookie", [`transactionQueryingToken=${authToken}`]);
    const dbItem = await prisma.Item.findUnique({
      where: {
        id: item.id,
      },
    });
    // try to get item from plaid
    const response = await getItemFromPlaid(item.accessToken);
    // clear items
    await clearItems();
    expect(dbItem).toEqual(null);
    expect(response.error_code).toEqual("ITEM_NOT_FOUND");
  });
});
