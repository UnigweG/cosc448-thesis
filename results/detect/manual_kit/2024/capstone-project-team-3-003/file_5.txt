import { Prisma } from "@prisma/client";
import type { Request, Response } from "express";
import { v4 as uuidv4 } from "uuid";
import type { SignUpRequest } from "../types/types";
import { Emails } from "../utilities/email";
import { getItemsForUser, removeItem } from "./dbPlaid";
const prisma = require("../../prisma/prisma");
const bcrypt = require("bcryptjs");
const jwt = require("jsonwebtoken");

const salt = process.env.SALT_ROUNDS
  ? Number.parseInt(process.env.SALT_ROUNDS)
  : 10;

/**
 * Validates request body. Hashes password. Stores user info in database.
 * @param req
 * @param res
 * @returns
 */
export async function signUp(req: Request, res: Response) {
  try {
    const errors = validateInput(req.body);
    if (Object.keys(errors).length) {
      return res.status(422).json({ errors: errors });
    }
    const { email, password, name } = req.body;
    const hash = bcrypt.hashSync(password, salt);
    const user = await prisma.User.create({
      data: {
        name,
        email,
        password: hash,
      },
    });
    const emails = new Emails();
    emails.sendConfirmation(email, user.uuid);
    return res.status(201).json({ response: "Account Created" });
  } catch (e) {
    if (e instanceof Prisma.PrismaClientKnownRequestError) {
      if (e.code === "P2002") {
        const error = "Breaks unique key constraint.";
        console.error(error);
        return res.status(422).json({ error: error });
      }
    }
    return res.status(400).json({ error: e });
  }
}

/**
 * Deletes one's own user account if password matches database credentials.
 * @param req
 * @param res
 * @returns
 */
export async function deleteUser(req: Request, res: Response) {
  try {
    const errors = validateInput(req.body);
    if (Object.keys(errors).length) {
      return res.status(422).json({ errors: errors });
    }
    const { password } = req.body;
    const user = await prisma.User.findUnique({
      where: {
        email: req.currentUser!.email,
      },
      select: {
        id: true,
        password: true,
      },
    });
    if (!user) {
      return res.status(422).json({ error: "Email not in database." });
    }
    const { id, password: hashedPassword } = user;
    if (!bcrypt.compareSync(password, hashedPassword)) {
      return res.status(401).json({ error: "Incorrect password." });
    }

    const items = await getItemsForUser(id);
    // remove items for user first
    await Promise.all(
      items.map(async (item: any) => {
        return await removeItem(item.accessToken);
      }),
    );
    const deleteUser = await prisma.User.delete({
      where: {
        email: req.currentUser!.email,
      },
    });
    if (!deleteUser) {
      return res.status(400).send("Error deleting account");
    }
    return res.status(200).send("Account deleted");
  } catch (error) {
    console.error(`Error: ${error}`);
    res.status(400).json({ error: error });
  }
}

/**
 * Logs in a user. Validates credentials. Signs a jwt token and sends as a cookie. If user has not confirmed their email, deny access.
 * @param req
 * @param res
 * @returns
 */
export async function login(req: Request, res: Response) {
  try {
    const errors = validateInput(req.body);
    if (Object.keys(errors).length) {
      return res.status(422).json({ error: errors });
    }
    const { email, password } = req.body;
    const user = await prisma.User.findUnique({
      where: {
        email,
      },
    });
    if (!user) {
      return res.status(422).json({ error: "Invalid credentials" });
    }
    const databasePassword = user.password;
    if (!bcrypt.compareSync(password, databasePassword)) {
      return res.status(422).json({ error: "Invalid credentials" });
    }
    if (!user.confirmed) {
      return res.status(401).json({ error: "Must verify email" });
    }
    // Create token that expires in an hour
    const token = jwt.sign({ uuid: user.uuid }, process.env.SECRET_JWT_KEY, {
      expiresIn: "1h",
    });
    res.cookie("transactionQueryingToken", token, {
      httpOnly: true,
      secure: true,
    });
    res.status(200).json({ token });
  } catch (error) {
    console.error(`Error: ${error}`);
    res.status(400).json({ error: error });
  }
}
